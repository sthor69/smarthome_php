<script>
    // Verifica autenticazione
    fetch('auth.php?action=status')
        .then(r => r.json())
        .then(data => {
            if (!data.logged_in) {
                window.location.href = 'login.html';
            }
        })
        .catch(() => { window.location.href = 'login.html'; });

    function logout() {
        fetch('auth.php?action=logout')
            .then(() => { window.location.href = 'login.html'; });
    }

    let sensorChart;
    let currentHours = 6;
    let config = {
        discardThreshold: 10,
        movingAverageWindow: 5,
        enableInterpolation: false,
        polyDegree: 3,
        polyLambda: 0.01
    };

    function getCurrentRangeHours() {
        const startVal = document.getElementById('startDate').value;
        const endVal = document.getElementById('endDate').value;
        if (startVal && endVal) {
            const startD = new Date(startVal);
            const endD = new Date(endVal);
            return (endD - startD) / (1000 * 60 * 60);
        }
        return currentHours;
    }

    function getMaxPolyDegree(hours) {
        if (hours <= 0) return 20; // "Tutto"
        if (hours <= 1) return 9;
        if (hours <= 6) return 12;
        if (hours <= 24) return 15;
        if (hours <= 168) return 18;
        return 20;
    }

    function loadConfig() {
        const saved = localStorage.getItem('sensorConfig');
        if (saved) {
            config = Object.assign(config, JSON.parse(saved));
        }
        document.getElementById('discardThreshold').value = config.discardThreshold;
        document.getElementById('movingAverageWindow').value = config.movingAverageWindow;
        document.getElementById('enableInterpolation').checked = !!config.enableInterpolation;
        document.getElementById('polyDegree').value = config.polyDegree;
        document.getElementById('polyLambda').value = config.polyLambda || 0.01;
        togglePolyDegreeVisibility();
    }

    function togglePolyDegreeVisibility() {
        const enabled = document.getElementById('enableInterpolation').checked;
        document.getElementById('polyDegreeGroup').style.display = enabled ? 'block' : 'none';
    }

    document.getElementById('enableInterpolation')?.addEventListener('change', togglePolyDegreeVisibility);

    function openConfig() {
        const rangeHours = getCurrentRangeHours();
        const maxDegree = getMaxPolyDegree(rangeHours);
        const polyInput = document.getElementById('polyDegree');
        const polyHelp = document.getElementById('polyDegreeHelp');

        polyInput.max = maxDegree;
        if (parseInt(polyInput.value) > maxDegree) {
            polyInput.value = maxDegree;
        }
        polyHelp.textContent = `Grado della curva di regressione polinomiale (max ${maxDegree} per l'intervallo attuale).`;

        document.getElementById('configModal').style.display = 'flex';
    }

    function closeConfig() {
        document.getElementById('configModal').style.display = 'none';
    }

    function saveSettings() {
        const threshold = parseInt(document.getElementById('discardThreshold').value);
        const windowSize = parseInt(document.getElementById('movingAverageWindow').value);
        const polyDeg = parseInt(document.getElementById('polyDegree').value);
        const polyLamb = parseFloat(document.getElementById('polyLambda').value);
        const enabled = document.getElementById('enableInterpolation').checked;

        const rangeHours = getCurrentRangeHours();
        const maxDegree = getMaxPolyDegree(rangeHours);

        if (isNaN(threshold) || threshold < 0 || threshold > 100) {
            alert('Inserisci una soglia di scarto valida (0-100).');
            return;
        }
        if (isNaN(windowSize) || windowSize < 1 || windowSize > 50) {
            alert('Inserisci una finestra di media mobile valida (1-50).');
            return;
        }
        if (enabled && (isNaN(polyDeg) || polyDeg < 1 || polyDeg > maxDegree)) {
            alert(`Inserisci un grado del polinomio valido (1-${maxDegree}).`);
            return;
        }
        if (enabled && (isNaN(polyLamb) || polyLamb < 0)) {
            alert('Inserisci un valore di regolarizzazione valido.');
            return;
        }

        config.discardThreshold = threshold;
        config.movingAverageWindow = windowSize;
        config.enableInterpolation = enabled;
        config.polyDegree = polyDeg;
        config.polyLambda = polyLamb;

        localStorage.setItem('sensorConfig', JSON.stringify(config));
        closeConfig();
        updateChart();
    }

    window.onclick = function(event) {
        const modal = document.getElementById('configModal');
        if (event.target == modal) {
            closeConfig();
        }
    }

    loadConfig();

    const periodBtns = {
        1: 'btn1', 6: 'btn6', 24: 'btn24',
        168: 'btn168', 720: 'btn720', 0: 'btn0'
    };

    function roundTo5Minutes(d) {
        d.setMinutes(Math.round(d.getMinutes() / 5) * 5);
        d.setSeconds(0);
        d.setMilliseconds(0);
        return d;
    }

    function formatDateForInput(d) {
        const y = d.getFullYear();
        const m = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        const h = String(d.getHours()).padStart(2, '0');
        const min = String(d.getMinutes()).padStart(2, '0');
        return `${y}-${m}-${day}T${h}:${min}`;
    }

    function adjustTime(id, deltaMinutes) {
        const el = document.getElementById(id);
        let val = el.value;
        let d;

        if (val) {
            d = new Date(val);
        } else {
            d = new Date();
        }

        d.setMinutes(d.getMinutes() + deltaMinutes);
        roundTo5Minutes(d);
        el.value = formatDateForInput(d);
    }

    document.getElementById('startDate').addEventListener('change', function() {
        if (this.value) {
            const d = new Date(this.value);
            roundTo5Minutes(d);
            this.value = formatDateForInput(d);
        }
    });

    document.getElementById('endDate').addEventListener('change', function() {
        if (this.value) {
            const d = new Date(this.value);
            roundTo5Minutes(d);
            this.value = formatDateForInput(d);
        }
    });

    ['startDate', 'endDate'].forEach(id => {
        document.getElementById(id).addEventListener('wheel', function(e) {
            e.preventDefault();
            const delta = e.deltaY < 0 ? 15 : -15;
            adjustTime(id, delta);
        });
    });

    function setPeriod(hours) {
        document.getElementById('startDate').value = '';
        document.getElementById('endDate').value = '';
        currentHours = hours;
        Object.entries(periodBtns).forEach(([h, id]) => {
            document.getElementById(id).classList.toggle('active', parseInt(h) === hours);
        });
        updateChart();
    }

    function applyCustomRange() {
        const startVal = document.getElementById('startDate').value;
        const endVal = document.getElementById('endDate').value;

        if (!startVal || !endVal) {
            alert('Inserisci sia la data di inizio che quella di fine.');
            return;
        }

        // Disattiva pulsanti preset
        Object.values(periodBtns).forEach(id => {
            document.getElementById(id).classList.remove('active');
        });

        updateChart();
    }

    function calculateSlope(data) {
        if (data.length < 2) return 0;

        // Prendiamo i dati delle ultime 6 ore per il trend (se disponibili)
        // Altrimenti usiamo tutto quello che abbiamo
        const now = data[data.length - 1].x.getTime();
        const sixHoursAgo = now - (6 * 3600 * 1000);
        const recentData = data.filter(p => p.x.getTime() >= sixHoursAgo && p.y !== null);
